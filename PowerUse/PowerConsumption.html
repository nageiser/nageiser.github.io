Define prepare_features to format test and train data sets and one-hot encode categorical variables for the gradient-boosted tree:

prepare_features <- function(df, fmla) {
  df <- df %>%
    select(-year) %>%
    mutate(date = as.Date(date, format = "%m/%d/%Y")) %>%
    mutate(across(c(semester, quarter, day_in_week, week_in_year, day_in_year, month), as.factor))
  encoded_df <- as.data.frame(model.matrix(fmla, data = df))
  df <- df %>%
    bind_cols(encoded_df)
  return(df)
}
